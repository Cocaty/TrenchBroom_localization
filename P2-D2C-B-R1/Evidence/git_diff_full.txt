+// Install the QTranslator for the user's interface language (see reports/P2-D2C-A).
+// Must run after PreferenceManager is initialized and before any UI is created.
+// Never blocks or fails startup: any load failure just means the UI stays English.
+void initTranslator(QApplication& app)
+{
+  const auto& language = pref(Preferences::InterfaceLanguage);
+  if (language != "zh_CN")
+  {
+    return; // "en" (default) and any unknown value stay English
+  }
+  auto* translator = new QTranslator(&app); // parented to app: lives for the whole application
+  if (translator->load(":/translations/TrenchBroom_zh_CN.qm"))
+  {
+    app.installTranslator(translator);
+  }
+  else
+  {
+    qWarning() << "Could not load Chinese translation from :/translations/TrenchBroom_zh_CN.qm; "
+                  "falling back to English";
+    translator->deleteLater();
+  }
+}
